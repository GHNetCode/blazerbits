📦 Product API · Express.js Backend

Fully featured REST API with persistent JSON storage, auto‑increment IDs, advanced filtering, and bulk operations. Perfect for frontend dashboards & portfolio demos.

Node.js + Express 5 File‑based DB (db.json) CRUD + Bulk Dynamic data.* filtering Postman ready

📌 Project overview

This API manages a collection of products with flexible metadata. All data lives in db.json and is updated atomically. Built for simplicity, it powers full CRUD operations, smart filtering, and includes a bulk creation endpoint that assigns sequential numeric IDs. The server runs on http://localhost:3000.

✨ Auto‑increment IDs Numeric IDs only — both single and bulk POST operations guarantee consistent increments.
🔍 Deep filtering Use ?data.color=Cloudy%20White to filter by any first‑level property inside data object.
⚡ Bulk endpoint POST /products/bulk with array of products, all get new sequential IDs in one request.

Step 4 — Create server file (app.js)

Place the complete Express application in app.js. It handles all endpoints, filtering logic, and persistent writes.

📜 Full app.js code (copy exactly) 
const express = require('express');
const fs = require('fs').promises;
const path = require('path');
const app = express();
app.use(express.json());
const DB_PATH = path.join(__dirname, 'db.json');

async function readDB() { const data = await fs.readFile(DB_PATH, 'utf8'); return JSON.parse(data); }
async function writeDB(data) { await fs.writeFile(DB_PATH, JSON.stringify(data, null, 2)); }
async function getNextId() { const db = await readDB(); const maxId = db.products.reduce((max, p) => Math.max(max, Number(p.id)), 0); return maxId + 1; }

// GET /products with advanced filtering (id list + data.* fields)
app.get('/products', async (req, res) => {
  const db = await readDB();
  let results = db.products;
  if (req.query.id) {
    let ids = Array.isArray(req.query.id) ? req.query.id.map(Number) : req.query.id.split(',').map(Number);
    results = results.filter(p => ids.includes(Number(p.id)));
  }
  const dataFilters = Object.keys(req.query).filter(key => key.startsWith('data.'));
  for (const key of dataFilters) {
    const field = key.substring(5);
    const expectedValue = req.query[key];
    results = results.filter(item => item.data && typeof item.data === 'object' && item.data[field] == expectedValue);
  }
  res.json(results);
});

app.get('/products/:id', async (req, res) => {
  const db = await readDB();
  const product = db.products.find(p => p.id === Number(req.params.id));
  if (!product) return res.status(404).json({ error: 'Product not found' });
  res.json(product);
});

app.post('/products', async (req, res) => {
  const db = await readDB();
  const newId = await getNextId();
  const newProduct = { id: newId, ...req.body };
  newProduct.id = newId;
  db.products.push(newProduct);
  await writeDB(db);
  res.status(201).json(newProduct);
});

app.post('/products/bulk', async (req, res) => {
  const db = await readDB();
  if (!Array.isArray(req.body)) return res.status(400).json({ error: 'Request body must be an array of products' });
  const currentMax = db.products.reduce((max, p) => Math.max(max, Number(p.id)), 0);
  let nextId = currentMax + 1;
  const addedProducts = [];
  for (const product of req.body) {
    const newProduct = { id: nextId, ...product };
    newProduct.id = nextId;
    db.products.push(newProduct);
    addedProducts.push(newProduct);
    nextId++;
  }
  await writeDB(db);
  res.status(201).json(addedProducts);
});

app.put('/products/:id', async (req, res) => {
  const db = await readDB();
  const id = Number(req.params.id);
  const index = db.products.findIndex(p => p.id === id);
  if (index === -1) return res.status(404).json({ error: 'Product not found' });
  const updatedProduct = { id, ...req.body };
  updatedProduct.id = id;
  db.products[index] = updatedProduct;
  await writeDB(db);
  res.json(updatedProduct);
});

app.patch('/products/:id', async (req, res) => {
  const db = await readDB();
  const id = Number(req.params.id);
  const index = db.products.findIndex(p => p.id === id);
  if (index === -1) return res.status(404).json({ error: 'Product not found' });
  db.products[index] = { ...db.products[index], ...req.body, id };
  await writeDB(db);
  res.json(db.products[index]);
});

app.delete('/products/:id', async (req, res) => {
  const db = await readDB();
  const id = Number(req.params.id);
  const index = db.products.findIndex(p => p.id === id);
  if (index === -1) return res.status(404).json({ error: 'Product not found' });
  db.products.splice(index, 1);
  await writeDB(db);
  res.status(204).send();
});

const PORT = 3000;
app.listen(PORT, () => console.log(`✅ Server running on http://localhost:${PORT}`));

📡 REST Endpoints & usage

MethodEndpointDescription
GET/productsList all products. Supports ?id=1,2,5 and ?data.color=Cloudy%20White filtering.
GET/products/:idGet single product by numeric ID.
POST/productsCreate one product → auto‑incremented ID.
POST/products/bulkAccepts array of products, assigns sequential IDs starting from current max+1.
PUT/products/:idFull replace of product (excluding ID override).
PATCH/products/:idPartial update.
DELETE/products/:idRemove product → 204 No Content.

🐧 Postman setup (Linux — manual install + desktop icon)

If you’re on Ubuntu and the App Center version has UI glitches, use the official tarball and then ensure the desktop icon works correctly.

cd ~/Downloads
wget https://dl.pstmn.io/download/latest/linux64 -O postman-linux-x64.tar.gz   # or download from official site
tar -xzf postman-linux-x64.tar.gz
sudo mv Postman /opt/
sudo ln -s /opt/Postman/Postman /usr/local/bin/postman

# Create desktop entry
sudo bash -c 'cat > /usr/share/applications/postman.desktop << EOF
[Desktop Entry]
Name=Postman
Exec=/opt/Postman/Postman --ozone-platform=x11
Icon=/opt/Postman/app/resources/app/assets/icon.png
Terminal=false
Type=Application
Categories=Development;
EOF'

# IMPORTANT: Make the desktop file executable and mark it as trusted
chmod +x ~/Desktop/postman.desktop
gio set ~/Desktop/postman.desktop metadata::trusted true

sudo update-desktop-database
postman   # launch

After these steps, the Postman icon will appear on your desktop and in the app menu without any “untrusted” warning.

Once Postman is open, create a new request → set method to POST/GET/DELETE and use http://localhost:3000 endpoints.

📌 Key design notes

  • Numeric IDs only — both single and bulk endpoints ensure IDs are numbers and strictly sequential.
  • Data filter flexibility: data.* query parameters match top-level keys inside product.data (primitive values: string, number).
  • Bulk performance — reads max ID once, then assigns IDs in memory for efficiency.
  • Persistent storage — every POST, PUT, PATCH, DELETE rewrites db.json atomically.
  • Graceful error handling — 404 for missing products, 400 for malformed bulk requests.
